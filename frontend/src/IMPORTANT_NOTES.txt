CREATE TABLE servers (
    id SERIAL PRIMARY KEY,
    server_name VARCHAR(99) NOT NULL,
    server_icon VARCHAR(999) NOT NULL,
    server_id VARCHAR(999) UNIQUE NOT NULL,
    server_owner VARCHAR(99) NOT NULL,
    server_description VARCHAR(500),
    server_thumbnail VARCHAR(999),
    server_channels INTEGER DEFAULT 0,
    server_channel_array TEXT[] NOT NULL DEFAULT '{}',
    server_roles_array TEXT[] NOT NULL DEFAULT '{}',
    server_roles INTEGER DEFAULT 0,
    server_members_array TEXT[] NOT NULL DEFAULT '{}',
    server_members_banned_array TEXT[] NOT NULL DEFAULT '{}',
);

CREATE TABLE member_roles (
    id SERIAL PRIMARY KEY,
    member_roles_server_id VARCHAR(999) UNIQUE NOT NULL,
    member_roles_username VARCHAR(999) UNIQUE NOT NULL,
    member_roles_array TEXT[] NOT NULL DEFAULT '{}'
);

CREATE TABLE channels (
    id SERIAL PRIMARY KEY,
    channel_id VARCHAR(999) UNIQUE NOT NULL,
    channel_name VARCHAR(99) NOT NULL,
    channel_description VARCHAR(99) NOT NULL,
);

CREATE TABLE messages (
    id SERIAL PRIMARY KEY,
    messages_channel_id VARCHAR(999) NOT NULL,
    messages_username VARCHAR(99) NOT NULL,
    messages_message TEXT, <---- THIS SUPPORTS EMOJIS
    messages_created_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE direct_messages (
    id SERIAL PRIMARY KEY,
    direct_messages_id VARCHAR (999) NOT NULL,
    direct_messages_username VARCHAR(99) NOT NULL,
    direct_messages_message TEXT,
    direct_messages_create_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE roles (
    id SERIAL PRIMARY KEY,
    role_id VARCHAR(999) UNIQUE NOT NULL,
    role_server_id VARCHAR(999) NOT NULL,
    role_name VARCHAR(99) NOT NULL,
    role_rank INTEGER NOT NULL DEFAULT 0,
    role_color VARCHAR(7) NOT NULL,
    can_kick_lower_rank_members BOOLEAN NOT NULL DEFAULT FALSE,
    can_ban_lower_rank_members BOOLEAN NOT NULL DEFAULT FALSE,
    can_edit_lower_rank_member_roles BOOLEAN NOT NULL DEFAULT FALSE
);
